baudrate=115200
bootdelay=1
ethaddr=ba:d0:4a:9c:4e:ce
stderr=serial
stdin=serial
stdout=serial
mtdparts=mtdparts=nand0:0x80000(u-boot),0x80000(env),0x80000(spl),0x80000(dtb),0x1400000(kernel),-(user)
loadkernel=nand read 0x7fc0 0x200000 0x1400000
loaddtb=nand read 0x1400000 0x180000 0x80000
bootargs=root=/dev/ram0 console=ttyS0,115200n8 mem=64M ${mtdparts} rootwait rw
bootcmd=run loadkernel;run loaddtb;bootm 0x7fc0 - 0x1400000
